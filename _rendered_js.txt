    let selectedTagCodes = JSON.parse(tagCodesInput.value || '[]');

    // Store tag metadata for display
    const tagMetaMap = {};  // code -> {code, name}

    const tagState = {
      page: 1,
      perPage: 10,
      total: 0,
      totalPages: 1,
      items: [],
      loading: false,

    function updateTagDisplay() {
      tagCodesInput.value = JSON.stringify(selectedTagCodes);
      if (selectedTagCodes.length === 0) {
        tagSelectionText.textContent = '当前：不筛选';
      } else {
        const names = selectedTagCodes.map(code => {
          const meta = tagMetaMap[code];
          return meta ? `${meta.name} (${meta.code})` : code;
        });
        tagSelectionText.textContent = `已选 ${selectedTagCodes.length} 个标签`;
      }
      tagClearBtn.disabled = selectedTagCodes.length === 0;

      // Render chips
      tagChipsContainer.innerHTML = '';
      selectedTagCodes.forEach(code => {
        const meta = tagMetaMap[code];
        const chip = document.createElement('div');
        chip.className = 'tag-chip';
        const label = document.createElement('span');
        label.textContent = meta ? `${meta.name}` : code;
        const removeBtn = document.createElement('button');
        removeBtn.className = 'tag-chip-remove';
        removeBtn.textContent = '×';
        removeBtn.onclick = () => {
          selectedTagCodes = selectedTagCodes.filter(c => c !== code);
          tagSelectionDirty = true;
          updateTagDisplay();
          renderTags();
          queueQuickConfigSave();
        };
        chip.appendChild(label);
        chip.appendChild(removeBtn);
        tagChipsContainer.appendChild(chip);
      });
    }

    function openTagModal() {
      tagModal.classList.add('show');
      tagModal.setAttribute('aria-hidden', 'false');
      document.body.classList.add('modal-open');
      if (!tagState.items.length && !tagState.loading) {
        loadTags(tagState.page);
      } else {
        renderTags();
      }
    }

    function closeTagModal() {
      tagModal.classList.remove('show');
      tagModal.setAttribute('aria-hidden', 'true');
      document.body.classList.remove('modal-open');
    }

    function handleTagModalBackdrop(event) {
      if (event.target === tagModal) {
        closeTagModal();
      }
    }

    function clearAllTags() {
      selectedTagCodes = [];
      tagSelectionDirty = true;
      updateTagDisplay();
      renderTags();
      queueQuickConfigSave();
    }

    function renderTagPagination() {
      tagPageInfo.textContent = `第 ${tagState.page} / ${tagState.totalPages} 页，共 ${tagState.total} 个标签`;
      tagPrevBtn.disabled = tagState.loading || tagState.page <= 1;
      tagNextBtn.disabled = tagState.loading || tagState.page >= tagState.totalPages;
    }

    function renderTags() {
      tagList.innerHTML = '';

      if (!tagState.items.length) {
        const empty = document.createElement('div');
        empty.className = 'note';
        empty.textContent = tagState.loading ? '正在加载标签...' : '暂无可用标签';
        tagList.appendChild(empty);
        renderTagPagination();
        return;
      }

      tagState.items.forEach((tag) => {
        const button = document.createElement('button');
        button.type = 'button';
        const isActive = selectedTagCodes.includes(tag.code);
        button.className = `tag-item${isActive ? ' active' : ''}`;
        button.disabled = tagState.loading;

        // Check mark
        const check = document.createElement('span');
        check.className = 'tag-check';
        check.textContent = isActive ? '✓' : '';
        button.appendChild(check);

        const name = document.createElement('div');
        name.className = 'tag-name';
        name.textContent = tag.name || tag.code;

        const code = document.createElement('div');
        code.className = 'tag-code';
        code.textContent = tag.code;

        button.appendChild(name);
        button.appendChild(code);

        button.addEventListener('click', () => {
          if (selectedTagCodes.includes(tag.code)) {
            selectedTagCodes = selectedTagCodes.filter(c => c !== tag.code);
          } else {
            selectedTagCodes.push(tag.code);
          }
          tagSelectionDirty = true;
          tagMetaMap[tag.code] = tag;
          updateTagDisplay();
          renderTags();
          queueQuickConfigSave();
        });

        tagList.appendChild(button);
      });

      renderTagPagination();
      updateTagDisplay();
    }

    async function loadTags(page = 1) {
      const targetPage = Math.max(1, page);
      tagState.loading = true;
      tagError.textContent = '';
      renderTags();

      try {
        const resp = await fetch(`/api/tags?page=${targetPage}&per_page=${tagState.perPage}`);
        const data = await resp.json();
        if (!data.ok) {
          throw new Error(data.error || '标签加载失败');
        }

        tagState.page = data.pagination?.page || 1;
        tagState.total = data.pagination?.total || 0;
        tagState.totalPages = data.pagination?.total_pages || 1;
        tagState.items = Array.isArray(data.items) ? data.items : [];

        // Store metadata for all loaded tags
        tagState.items.forEach(tag => {
          tagMetaMap[tag.code] = tag;
        });
      } catch (error) {
        tagState.items = [];
        tagState.total = 0;
        tagState.totalPages = 1;
        tagState.page = 1;
        tagError.textContent = error.message || '标签加载失败';
      } finally {
        tagState.loading = false;
        renderTags();
      }
    }

    // Pre-load all tag metadata for chip display
    async function preloadAllTagNames() {
      try {
        const resp = await fetch(`/api/tags?page=1&per_page=30`);
        const data = await resp.json();
        if (data.ok && data.items) {
          data.items.forEach(tag => { tagMetaMap[tag.code] = tag; });
        }
        // Load remaining pages if needed
        if (data.pagination && data.pagination.total_pages > 1) {
          for (let p = 2; p <= data.pagination.total_pages; p++) {
            const resp2 = await fetch(`/api/tags?page=${p}&per_page=30`);
            const data2 = await resp2.json();
            if (data2.ok && data2.items) {
              data2.items.forEach(tag => { tagMetaMap[tag.code] = tag; });
            }
          }
        }
        updateTagDisplay();
      } catch (e) {
        // ignore
      }
    }

    function setTimeFilterValue(minTime, maxTime) {
      const select = document.getElementById('settings_time_filter');
      const value = `${parseInt(minTime || 0)}-${parseInt(maxTime || 0)}`;
      if ([...select.options].some(option => option.value === value)) {
        select.value = value;
      } else {
        select.value = '0-86400';
      }
    }

      const option = document.getElementById('settings_time_filter').selectedOptions[0];
      return {
        min: parseInt(option.dataset.min || '0'),
        max: parseInt(option.dataset.max || '86400'),
      };
    }

    function sameTagCodes(left, right) {
      return JSON.stringify(left || []) === JSON.stringify(right || []);
    }

    async function saveQuickConfigNow() {
      if (!downloadRootDirty && !tagSelectionDirty) {
        return true;
      }

      const downloadRootSnapshot = downloadRootInput.value;
      const tagCodesSnapshot = [...selectedTagCodes];
      const formData = new FormData();
      formData.append('download_root', downloadRootSnapshot);
      formData.append('tag_codes', JSON.stringify(tagCodesSnapshot));

      try {
        const resp = await fetch('/save-quick', { method: 'POST', body: formData });
        const data = await resp.json();
        if (!data.ok) {
          throw new Error(data.error || '自动保存失败');
        }

        if (data.config) {
          currentConfig = data.config;
        }
        if (downloadRootInput.value === downloadRootSnapshot) {
          downloadRootDirty = false;
        }
        if (sameTagCodes(selectedTagCodes, tagCodesSnapshot)) {
          tagSelectionDirty = false;
        }
        return true;
      } catch (error) {
        console.error(error);
        return false;
      }
    }

    function queueQuickConfigSave(delay = 0) {
      window.clearTimeout(quickSaveTimer);
      quickSaveTimer = window.setTimeout(() => {
        quickSaveChain = quickSaveChain.then(saveQuickConfigNow);
      }, delay);
    }

    async function flushQuickConfigSave() {
      window.clearTimeout(quickSaveTimer);
      quickSaveChain = quickSaveChain.then(saveQuickConfigNow);
      return quickSaveChain;
    }

    downloadRootInput.addEventListener('input', () => {

      downloadRootDirty = true;
      currentConfig.download_root = downloadRootInput.value;

    async function runNow() {
      const quickSaved = await flushQuickConfigSave();
      if (!quickSaved) {
        alert('下载根目录或标签筛选自动保存失败，请检查后重试');
        return;

        alert(data.message || '启动失败');

    async function refreshStatus() {
      const resp = await fetch('/status');
      const data = await resp.json();
      if (!data.ok) {
        return;
      }

      document.getElementById('auto_download_enabled').innerText = data.config?.auto_download_enabled === false ? '已关闭' : '已开启';
      document.getElementById('is_running').innerText = data.state.is_running ? '是' : '否';
      document.getElementById('last_run_time').innerText = data.state.last_run_time || '暂无';
      document.getElementById('last_result').innerText = data.state.last_result || '';
      document.getElementById('logs').innerText = (data.logs || []).join('\n');
      // Update local config cache from server
      if (data.config) {
        currentConfig = data.config;
        if (!downloadRootDirty) {
          downloadRootInput.value = currentConfig.download_root || '';
        }
        // Update tag selection if config changed
        if (!tagSelectionDirty && Array.isArray(currentConfig.tag_codes)) {
          selectedTagCodes = [...currentConfig.tag_codes];
          updateTagDisplay();
        }
      }
      // Update blogger state display
      if (data.blogger_state) {
        document.getElementById('blogger_is_running').textContent = data.blogger_state.is_running ? '是' : '否';
        document.getElementById('blogger_last_run_time').textContent = data.blogger_state.last_run_time || '暂无';
        document.getElementById('blogger_last_result').textContent = data.blogger_state.last_result || '';
      }
    }

    async function addBlogger() {
      const val = bloggerAddInput.value.trim().replace(/^@/, '').toLowerCase();
      if (!val) return;
      if (!/^[a-zA-Z0-9_]+$/.test(val)) {
        alert('用户名只能包含字母、数字和下划线');
        return;
      }
      try {
        const resp = await fetch('/api/blogger/add', {
          method: 'POST',
          headers: {'Content-Type': 'application/json'},
          body: JSON.stringify({screen_name: val}),
        });
        const data = await resp.json();
        if (!data.ok) {
          alert('添加失败：' + (data.error || '未知错误'));
          return;
        }
        bloggerAddInput.value = '';
        await refreshBloggerStatus();
      } catch (e) {
        alert('网络错误');
      }
    }

    async function removeBlogger(name) {
      if (!confirm('确定移除 @' + name + ' 吗？\n（已下载的文件不会被删除）')) return;
      try {
        const resp = await fetch('/api/blogger/remove', {
          method: 'POST',
          headers: {'Content-Type': 'application/json'},
          body: JSON.stringify({screen_name: name}),
        });
        const data = await resp.json();
        if (!data.ok) {
          alert('移除失败：' + (data.error || '未知错误'));
          return;
        }
        await refreshBloggerStatus();
      } catch (e) {
        alert('网络错误');
      }
    }
